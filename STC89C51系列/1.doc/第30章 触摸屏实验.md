# 第三十章 触摸屏实验（XPT2046/ADS7846 电阻屏，SPI）

## 1. 导入

上一章我们已驱动 ILI9341 TFT 彩屏完成图形与文字显示。本章在此基础上加入常见的电阻式触摸屏控制器 XPT2046（兼容 ADS7846），通过 4 线 SPI 实现触摸坐标读取与简单的人机交互（点按、按钮触发、画点）。

目标：

- 了解电阻屏 + XPT2046 的工作原理与引脚连接
- 完成 SPI 位操作读坐标（X/Y）、压力（Z）与去抖/滤波
- 将原始坐标映射到屏幕坐标（含方向与标定）
- 集成上一章 ILI9341 驱动，实现触摸画点与按钮点击

---

## 2. 硬件设计

- 屏幕与控制器
  - TFT：ILI9341（SPI 4 线）已接好
  - 触摸：XPT2046/ADS7846（与电阻屏玻璃连接）
- 供电与电平
  - 触摸芯片工作 3.3V 逻辑。若 MCU 为 5V（如 89C52），建议：
    - MOSI/SCK/CS/（可选 RST）降压到 3.3V（电阻分压或 74LVC 系列）
    - MISO 输入 MCU 一般可直接识别 3.3V 为高
  - 很多带 ILI9341 的 SPI 模块已板载电平转换，请确认丝印与说明
- 引脚连接（与 ILI9341 共用 SCK/MOSI，独立 CS/MISO/PENIRQ）
  - ILI9341（上一章示例）：
    - `P1.0 → CS_LCD`，`P1.1 → DC`，`P1.2 → RST_LCD`，`P1.3 → SCK`，`P1.4 → MOSI`
  - XPT2046（本章新增）：
    - `P1.5 → CS_T`（触摸片选，低有效）
    - `P1.6 ← MISO_T`（触摸数据输出）
    - `P1.7 ← PENIRQ`（触摸中断，按下为低，带上拉）
    - `P1.3 → SCK`（与 LCD 共用）
    - `P1.4 → MOSI`（与 LCD 共用）
  - GND 共地；VCC 3.3V
- 建议：PENIRQ 接上拉（10k 到 3.3V），并走短线；MISO 走线尽量短且靠地

## 3. 协议与命令速览（XPT2046）

- SPI，模式 0（CPOL=0, CPHA=0），MSB 先行
- 控制字（8 位）：
  - b7=1（Start）
  - b6..b4：通道选择 A2 A1 A0
    - X：`101`（0xD?），Y：`001`（0x9?），Z1：`011`（0xB?），Z2：`100`（0xC?）
  - b3：12 位/8 位模式（1=8bit，0=12bit）→ 选 0
  - b2：SER/DFR（0=差分，推荐；1=单端）
  - b1..b0：电源管理（常用 0b00：转换后省电；或 0b11：持续上电）
- 常用命令（12 位差分，省电模式）：
  - `CMD_X = 0xD0`，`CMD_Y = 0x90`，`CMD_Z1 = 0xB0`，`CMD_Z2 = 0xC0`
- 读取流程：
  - CS 置低 → 发送控制字 → 连续读两字节（丢弃头 4 位，取 12 位数值）→ CS 置高

---

## 4. 软件设计要点

- 同步与隔离
  - 与 LCD 共用 SCK/MOSI，但片选独立；任何时刻只能选中一个设备（CS）
- 触摸检测
  - PENIRQ 低表示“按下”；也可通过压力值（Z = Z2 - Z1）阈值判断
- 抗抖/滤波
  - 读取多次取中值或去极值平均
  - 快速滑动时建议快速采样减小延迟
- 坐标映射
  - 将原始（rawX/rawY）按 min/max 线性映射到屏幕像素坐标
  - 如方向不一致，可交换 X/Y 并翻转

## 5. 完整代码（Keil C51，触摸驱动 + 演示）

说明：

- 复用上一章 ILI9341 驱动，直接调用 `tft_init/tft_fill_screen/tft_draw_pixel/tft_fill_rect/tft_draw_rect/tft_print6x8` 等函数
- 若你的工程文件名不同，请相应修改 extern 声明或包含头文件

```c
#include <reg52.h>
#include <intrins.h>

/* ========== ILI9341 外部函数（请确保已添加上一章驱动文件） ========== */
extern void tft_init(void);
extern void tft_fill_screen(unsigned int color);
extern void tft_fill_rect(unsigned int x, unsigned int y, unsigned int w, unsigned int h, unsigned int color);
extern void tft_draw_rect(unsigned int x, unsigned int y, unsigned int w, unsigned int h, unsigned int color);
extern void tft_draw_pixel(unsigned int x, unsigned int y, unsigned int color);
extern void tft_draw_line(int x0,int y0,int x1,int y1,unsigned int color);
extern void tft_print6x8(unsigned int x, unsigned int y, const char* s, unsigned int fg, unsigned int bg);

/* ========== 与上一章一致的参数与颜色 ========== */
#define TFT_W 240
#define TFT_H 320
#define COLOR_BLACK   0x0000
#define COLOR_WHITE   0xFFFF
#define COLOR_RED     0xF800
#define COLOR_GREEN   0x07E0
#define COLOR_BLUE    0x001F
#define COLOR_YELLOW  0xFFE0
#define COLOR_CYAN    0x07FF
#define COLOR_MAGENTA 0xF81F
#define COLOR_GRAY    0x8410

/* ========== 触摸 XPT2046 引脚（按接线修改） ========== */
sbit T_CS   = P1^5;   // 触摸片选（低有效）
sbit T_MISO = P1^6;   // 触摸数据输出（MCU输入）
sbit T_IRQ  = P1^7;   // 触摸中断（按下=0）
/* 复用 LCD 的 SPI 时钟与数据输出 */
sbit SPI_SCK  = P1^3; // 共用
sbit SPI_MOSI = P1^4; // 共用

/* ========== 公共延时 ========== */
static void tiny_delay(void){ _nop_(); _nop_(); _nop_(); _nop_(); }
void delay_ms(unsigned int ms){ unsigned int i,j; for(i=0;i<ms;i++) for(j=0;j<125;j++); }

/* ========== XPT2046 控制字 ========== */
#define CMD_X   0xD0  // 1101 0000  (Start=1, A2..A0=101, 12bit, DFR, PD=00)
#define CMD_Y   0x90  // 1001 0000
#define CMD_Z1  0xB0  // 1011 0000
#define CMD_Z2  0xC0  // 1100 0000

/* ========== SPI 位操作：写8位并读8位（MISO） ========== */
static unsigned char spi_xfer8(unsigned char out){
    unsigned char i, in=0;
    for(i=0;i<8;i++){
        SPI_SCK = 0;
        SPI_MOSI = (out & 0x80) ? 1 : 0;
        tiny_delay();
        SPI_SCK = 1;
        in = (in << 1) | (T_MISO ? 1 : 0);
        out <<= 1;
        tiny_delay();
    }
    SPI_SCK = 0;
    return in;
}

/* 读取12位转换值：发控制字→读两字节→拼接并右移4位 */
static unsigned int xpt_read12(unsigned char cmd){
    unsigned char h, l;
    unsigned int val;

    T_CS = 0;
    spi_xfer8(cmd);      // 发送控制字
    h = spi_xfer8(0x00); // 高8位
    l = spi_xfer8(0x00); // 低8位
    T_CS = 1;

    val = ((unsigned int)h << 8) | l;
    val >>= 4;           // 12位有效
    return val;          // 0..4095
}

/* 读坐标原始值（丢弃首样本，取多次中值/均值） */
static unsigned int xpt_read_channel_avg(unsigned char cmd, unsigned char n){
    unsigned char i, j;
    unsigned int buf[7], tmp;
    if(n < 1) n = 1; if(n > 7) n = 7;

    // 丢弃首样本（参考做法，消除上一次控制字残留影响）
    (void)xpt_read12(cmd);

    for(i=0;i<n;i++){
        buf[i] = xpt_read12(cmd);
    }

    // 简易排序（升序），取中值；n为偶取中间靠前那个
    for(i=0;i<n;i++){
        for(j=i+1;j<n;j++){
            if(buf[j] < buf[i]){ tmp = buf[i]; buf[i] = buf[j]; buf[j] = tmp; }
        }
    }
    return buf[n/2];
}

/* 压力估算（Z=Z2-Z1，越大说明触摸越实） */
static unsigned int xpt_read_pressure(void){
    unsigned int z1 = xpt_read12(CMD_Z1);
    unsigned int z2 = xpt_read12(CMD_Z2);
    if(z2 > z1) return (z2 - z1);
    return 0;
}

/* ========== 触摸状态与坐标映射 ========== */
static bit touch_pressed_hw(void){
    // PENIRQ：按下为0；也可结合Z阈值做二次确认
    return (T_IRQ == 0) ? 1 : 0;
}

/* 标定参数（按你的屏与方向适配。先用估计值，实测后调整） */
#define RAW_X_MIN  300    // 触摸到左边缘时的原始X（实测校准）
#define RAW_X_MAX  3800   // 触摸到右边缘时的原始X
#define RAW_Y_MIN  300    // 顶边
#define RAW_Y_MAX  3800   // 底边

/* 若装配方向不同，可置1翻转/交换轴 */
#define FLIP_X     0
#define FLIP_Y     0
#define SWAP_XY    0

/* 将原始12位（0..4095）按标定映射到像素坐标 */
static bit map_raw_to_xy(unsigned int rx, unsigned int ry, unsigned int *px, unsigned int *py){
    long x = rx, y = ry;
#if SWAP_XY
    long t = x; x = y; y = t;
#endif
    if(x < RAW_X_MIN) x = RAW_X_MIN;
    if(x > RAW_X_MAX) x = RAW_X_MAX;
    if(y < RAW_Y_MIN) y = RAW_Y_MIN;
    if(y > RAW_Y_MAX) y = RAW_Y_MAX;

    x = (x - RAW_X_MIN) * (TFT_W - 1) / (RAW_X_MAX - RAW_X_MIN);
    y = (y - RAW_Y_MIN) * (TFT_H - 1) / (RAW_Y_MAX - RAW_Y_MIN);

#if FLIP_X
    x = (TFT_W - 1) - x;
#endif
#if FLIP_Y
    y = (TFT_H - 1) - y;
#endif
    *px = (unsigned int)x;
    *py = (unsigned int)y;
    return 1;
}

/* 读取一次有效坐标（含去抖）：返回1=有效，输出像素x,y */
static bit touch_read_point(unsigned int *x, unsigned int *y){
    unsigned int rx, ry, p;

    if(!touch_pressed_hw()) return 0;

    // 取多次中值，抑制抖动
    rx = xpt_read_channel_avg(CMD_X, 5);
    ry = xpt_read_channel_avg(CMD_Y, 5);

    // 压力简单筛选（可选）
    p  = xpt_read_pressure();
    if(p < 50) return 0;  // 阈值需根据屏与供电微调

    return map_raw_to_xy(rx, ry, x, y);
}

/* 等待松手（简易防抖） */
static void wait_release(void){
    unsigned int stable = 0;
    while(stable < 5){
        if(touch_pressed_hw()) stable = 0;
        else stable++;
        delay_ms(10);
    }
}

/* ========== 演示：画点 + 两个按钮点击 ========== */
static void ui_draw_buttons(void){
    // 左下角两个按钮
    tft_fill_rect(10,  TFT_H-50, 90, 36, COLOR_BLUE);
    tft_draw_rect(10,  TFT_H-50, 90, 36, COLOR_WHITE);
    tft_print6x8(20,   TFT_H-44, "CLEAR", COLOR_WHITE, COLOR_BLUE);

    tft_fill_rect(120, TFT_H-50, 90, 36, COLOR_GREEN);
    tft_draw_rect(120, TFT_H-50, 90, 36, COLOR_WHITE);
    tft_print6x8(130,  TFT_H-44, "BOX", COLOR_WHITE, COLOR_GREEN);
}

static bit in_rect(unsigned int x,unsigned int y,unsigned int rx,unsigned int ry,unsigned int w,unsigned int h){
    return (x>=rx && x<rx+w && y>=ry && y<ry+h) ? 1 : 0;
}

/* 主程序 */
void main(void){
    unsigned int x, y;
    unsigned int lastx=0xFFFF, lasty=0xFFFF;

    // 缺省电平
    T_CS   = 1;
    SPI_SCK= 0;
    SPI_MOSI = 0;

    tft_init();
    tft_fill_screen(COLOR_BLACK);
    tft_print6x8(8, 8, "Touch Demo (XPT2046)", COLOR_YELLOW, COLOR_BLACK);
    tft_print6x8(8, 24,"Tap to draw. Buttons below.", COLOR_WHITE, COLOR_BLACK);
    ui_draw_buttons();

    while(1){
        if(touch_read_point(&x,&y)){
            // 命中按钮？
            if(in_rect(x,y,10, TFT_H-50, 90,36)){
                // CLEAR
                tft_fill_rect(0, 40, TFT_W, TFT_H-60, COLOR_BLACK);
                ui_draw_buttons();
                wait_release();
                continue;
            }
            if(in_rect(x,y,120, TFT_H-50, 90,36)){
                // BOX
                tft_draw_rect(20, 60, TFT_W-40, TFT_H-120, COLOR_MAGENTA);
                wait_release();
                continue;
            }

            // 普通区域：画点/连线
            if(lastx!=0xFFFF){
                // 连线（画刷效果），注意性能
                tft_draw_line(lastx,lasty,x,y,COLOR_CYAN);
            }else{
                tft_draw_pixel(x,y,COLOR_CYAN);
            }
            lastx = x; lasty = y;
        }else{
            // 未按：复位“连线起点”
            lastx = lasty = 0xFFFF;
        }
    }
}
```

## 6. 标定与方向说明

- 初始 RAW_MIN/MAX 为估值。建议做一次“三点/四点”标定：
  1. 在屏四角（内缩 10px）依次画“十字”，提示用户点按
  2. 记录每次 rawX/rawY 与目标屏幕坐标
  3. 用线性拟合计算缩放与偏移（甚至带旋转项）
- 简化线性标定（无旋转）：按你实测的原始最小值/最大值替换 `RAW_X_MIN/MAX`、`RAW_Y_MIN/MAX` 即可
- 若坐标轴颠倒：
  - 交换 XY：`#define SWAP_XY 1`
  - 反转某轴：`#define FLIP_X 1` 或 `#define FLIP_Y 1`
- 与 ILI9341 显示方向相关：若调整了 `MADCTL(0x36)`，触摸映射也要同步修改

## 7. 测试与调试

- 若始终读不到按下：
  - 检查 PENIRQ 是否为低触发、是否上拉；CS 是否与 LCD 错开；电平转换是否可靠
- 坐标飘移/抖动大：
  - 提高采样次数 n（如 5→7），或采用“去极值平均”
  - 供电/参考稳定（屏背光噪声较大时可在 3.3V 侧加 10µF+0.1µF）
  - 缩短触摸信号线、加地护线
- 坐标方向错位：
  - 调整 FLIP/SWAP 或重新测量 RAW_MIN/MAX
- 按下变慢/卡顿：
  - 8051 位打 SPI 速率有限，减少每帧绘制像素量（尽量不全屏刷新）
  - 仅在按下时高频读取；松手时进入低功耗/低采样

## 8. 进阶与扩展

- 使用外设 SPI（若 MCU 支持）替代位操作，提升读写速度与稳定性
- 使用“持续上电”（PD=11）模式降低启动延时（功耗会上升）
- 加入“长按/双击/滑动”手势识别与简单 GUI 组件（按钮、滑条）
- 电容式触摸（如 FT6206 I²C）控制器：读坐标更稳、免标定，接线/驱动不同，可在 I²C 章节基础上扩展

## 9. 小结

- 硬件：XPT2046 与 ILI9341 共用 SPI 时钟/数据、独立 CS/MISO/PENIRQ；注意 3.3V 逻辑与地线质量
- 软件：完成 12 位坐标读取、压力判断、去抖/中值滤波与线性映射
- 应用：实现画点/连线与按钮点击，形成基础人机交互
- 优化：使用外设 SPI、减少刷新区域、合理标定与滤波提升体验
